local musicInputDefaults = setmetatable({}, { __mode = "k" })
local function musicInput(enabled)
    if not UI.musicPanel then return end
    local nodes = UI.musicPanel:GetDescendants()
    nodes[#nodes + 1] = UI.musicPanel
    for _, node in ipairs(nodes) do
        if node:IsA("GuiObject") then
            local defaults = musicInputDefaults[node]
            if not defaults then
                defaults = { active = node.Active }
                if node:IsA("ScrollingFrame") then defaults.scrolling = node.ScrollingEnabled end
                if node:IsA("TextBox") then defaults.editable = node.TextEditable end
                musicInputDefaults[node] = defaults
            end
            node.Active = enabled and defaults.active or false
            if node:IsA("GuiButton") then node.Interactable = enabled end
            if node:IsA("ScrollingFrame") then node.ScrollingEnabled = enabled and defaults.scrolling or false end
            if node:IsA("TextBox") then
                node.TextEditable = enabled and defaults.editable or false
                if not enabled then node:ReleaseFocus() end
            end
        end
    end
end

local function make(class, props, parent)
    local object = Instance.new(class)
    for key, value in pairs(props) do object[key] = value end
    object.Parent = parent
    return object
end
local function bind(signal, callback, bag)
    local connection = signal:Connect(callback)
    table.insert(bag or connections, connection)
    return connection
end
local function disconnect(bag)
    for _, connection in ipairs(bag) do connection:Disconnect() end
    table.clear(bag)
end
local function round(object, radius)
    make("UICorner", { CornerRadius = UDim.new(0, radius) }, object)
    return object
end
local function frame(parent, name, x, y, w, h, color, radius, alpha)
    local node = make("Frame", {
        Name = name, Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h),
        BackgroundColor3 = color or C.card, BackgroundTransparency = alpha or 0,
        BorderSizePixel = 0,
    }, parent)
    if radius then round(node, radius) end
    for _, key in ipairs({ "ink", "panel", "card", "white", "muted", "accent", "border", "icon" }) do
        if node.BackgroundColor3 == C[key] then themed(node, "BackgroundColor3", key); break end
    end
    return node
end
local function text(parent, value, x, y, w, h, size, color, bold)
    local node = make("TextLabel", {
        BackgroundTransparency = 1, Position = UDim2.fromOffset(x, y),
        Size = UDim2.fromOffset(w, h), Text = value, TextSize = math.max(13, size), TextTransparency = 0,
        TextColor3 = color or C.white, Font = bold and Enum.Font.GothamBold or Enum.Font.GothamSemibold,
        TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd,
    }, parent)
    for _, key in ipairs({ "white", "muted", "accent", "icon" }) do
        if node.TextColor3 == C[key] then themed(node, "TextColor3", key); break end
    end
    return node
end

local function gradient(object, top, bottom, rotation)
    return make("UIGradient", { Color = ColorSequence.new(top, bottom), Rotation = rotation or 90 }, object)
end
local function tween(object, props, duration, easing)
    if not alive then return end
    local bag = tracks[object] or {}
    for key in pairs(props) do if bag[key] then bag[key]:Cancel() end end
    tracks[object] = bag
    local animation = TweenService:Create(object, TweenInfo.new(state.motion and (duration or 0.35) or 0.01,
        easing or Enum.EasingStyle.Quint, Enum.EasingDirection.Out), props)
    for key in pairs(props) do bag[key] = animation end
    animation.Completed:Once(function()
        for key in pairs(props) do if bag[key] == animation then bag[key] = nil end end
        if next(bag) == nil and tracks[object] == bag then tracks[object] = nil end
    end)
    animation:Play()
    return animation
end
local function button(parent, value, x, y, w, h, light)
    local node = round(make("TextButton", {
        Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h),
        Text = value, TextSize = 16, TextColor3 = light and C.onAccent or C.white,
        Font = Enum.Font.GothamBold, BackgroundColor3 = light and C.accent or C.card, BackgroundTransparency = light and 0 or 0.32,
        BorderSizePixel = 0, AutoButtonColor = false,
    }, parent), 16)
    themed(node, "BackgroundColor3", light and "accent" or "card")
    themed(node, "TextColor3", light and "onAccent" or "white")
    local scale = make("UIScale", { Scale = 1 }, node)
    local baseAlpha = light and 0 or 0.32
    bind(node.MouseEnter, function()
        if node.BackgroundTransparency < 1 then tween(node, { BackgroundTransparency = math.max(0, baseAlpha - 0.14) }, 0.18) end
    end)
    bind(node.MouseLeave, function()
        if node.BackgroundTransparency < 1 then tween(node, { BackgroundTransparency = baseAlpha }, 0.18) end
        tween(scale, { Scale = 1 }, 0.48, Enum.EasingStyle.Elastic)
    end)
    bind(node.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            tween(scale, { Scale = 0.92 }, 0.12)
        end
    end)
    bind(node.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            tween(scale, { Scale = 1 }, 0.58, Enum.EasingStyle.Elastic)
        end
    end)
    return node
end

end
local function switch(parent, title, detail, y, getter, setter)
    local width = parent.Size.X.Offset
    text(parent, title, 24, y, width - 116, 29, 18, C.white, true)
    text(parent, detail, 24, y + 33, width - 48, 24, 13, C.muted)
    local hit = button(parent, "", width - 82, y + 6, 58, 34)
    round(hit, 24)
    local knob = frame(hit, "Thumb", 28, 4, 26, 26, Color3.new(1, 1, 1), 24)
    local function refresh()
        local on = getter()
        tween(hit, { BackgroundColor3 = on and C.accent or C.off }, 0.23)
        tween(knob, { Position = UDim2.fromOffset(on and 28 or 4, 4) }, 0.23)
    end
    bind(hit.Activated, function() setter(not getter()); refresh() end)
    switches[#switches + 1] = refresh
    refresh()
end
local function choice(parent, title, y, values, getter, setter)
    local available = parent.Size.X.Offset - 48
    text(parent, title, 24, y, available, 28, 18, C.white, true)
    local buttons = {}
    local function refresh()
        for i, node in ipairs(buttons) do
            tween(node, { BackgroundColor3 = getter() == i and C.accent or C.card,
                TextColor3 = getter() == i and C.onAccent or C.muted }, 0.25)
        end
    end
    for i, value in ipairs(values) do
        local width = (available - (#values - 1) * 8) / #values
        buttons[i] = button(parent, value, 24 + (i - 1) * (width + 8), y + 42, width, 40)
        buttons[i].TextSize = 14
        bind(buttons[i].Activated, function() setter(i); refresh() end)
    end
    switches[#switches + 1] = refresh
    refresh()
end

M.http = function(url, method, payload)
    if not alive then error("界面已关闭", 0) end
    if type(M.request) == "function" then
        local response = M.request({ Url = url, Method = method or "GET", Body = payload,
            Headers = { Referer = "https://music.163.com/", ["User-Agent"] = "Mozilla/5.0",
                ["Content-Type"] = "application/x-www-form-urlencoded" } })
        local status = type(response) == "table" and tonumber(response.StatusCode or response.Status)
        local body = type(response) == "table" and (response.Body or response.body)
        if status and (status < 200 or status >= 300) then error("HTTP " .. status, 0) end
        if type(body) == "string" and #body > 0 then return body end
        error("请求未返回有效内容", 0)
    end
    local ok, body = pcall(function() return game:HttpGet(url) end)
    if ok and type(body) == "string" and #body > 0 then return body end
    error("当前运行环境未提供 HTTP 请求能力", 0)
end
M.json = function(path, method, payload)
    local body = M.http(Config.MusicApi .. path, method, payload)

    if type(data) ~= "table" or data.code and tonumber(data.code) ~= 200 then error("网易云接口暂不可用", 0) end

end

    if type(M.write) ~= "function" or type(M.asset) ~= "function" then

    end
    local path = M.folder .. "aurora-" .. name
    M.write(path, bytes)
    return M.asset(path)
end

M.playSong = function(song, index)
    M.version = M.version + 1
    local version = M.version
    M.index = index or M.index
    M.loading, M.loaded, M.wants, M.ended = true, false, true, false
    M.sound:Stop()
    M.name, M.artistName, M.duration = song.name, song.artist, song.duration or 0
    M.title.Text, M.artist.Text, M.status.Text = M.name, M.artistName, "正在准备歌曲…"
    M.expanded = false
    M.cover.Image, M.miniCover.Image, M.detailCover.Image = "", "", ""
    M.update()
    task.spawn(function()
        local ok, err = pcall(function()
            if type(M.write) ~= "function" or type(M.asset) ~= "function" then error("播放需要 writefile 与 getcustomasset 支持", 0) end
            local detail = M.json("/api/song/detail/?id=" .. song.id .. "&ids=%5B" .. song.id .. "%5D")
            if not alive or version ~= M.version then return end
            if detail.songs and detail.songs[1] then song = M.normalize(detail.songs[1]) end
            M.name, M.artistName, M.duration = song.name, song.artist, song.duration
            M.title.Text, M.artist.Text = song.name, song.artist
            M.status.Text = "正在获取可播放音源…"
            local asset = M.cache[song.id]
            if not asset then
                local data = M.json("/api/song/enhance/player/url?ids=%5B" .. song.id .. "%5D&br=128000")
                if not alive or version ~= M.version then return end
                local entry = data.data and data.data[1]
                local url = entry and entry.url
                if type(url) ~= "string" or url == "" then url = "https://music.163.com/song/media/outer/url?id=" .. song.id .. ".mp3" end
                local bytes = M.http(url:gsub("^http:", "https:"))
                if not alive or version ~= M.version then return end
                local a, b = bytes:byte(1, 2)
                if #bytes < 1024 or not (bytes:sub(1, 3) == "ID3" or a == 255 and b and b >= 224) then
                    error("该歌曲暂未提供可播放的 MP3 音源", 0)
                end
                asset = M.save("song-" .. song.id .. ".mp3", bytes)
                M.cache[song.id] = asset
            end
            if not alive or version ~= M.version then return end
            M.sound.SoundId = asset
            local deadline = os.clock() + 15
            while alive and version == M.version and not M.sound.IsLoaded and os.clock() < deadline do task.wait(0.1) end
            if not alive or version ~= M.version then return end
            if not M.sound.IsLoaded then error("本地音频载入超时", 0) end
            M.loading, M.loaded = false, true
            M.sound:Play()
            if not M.wants then M.sound:Pause() end
            M.status.Text = M.wants and "正在播放" or "已暂停"
            M.update()
            if song.cover then
                task.spawn(function()
                    local success, cover = pcall(function()
                        local bytes = M.http(song.cover:gsub("^http:", "https:") .. (song.cover:find("?", 1, true) and "&" or "?") .. "param=256y256")
                        if not alive or version ~= M.version then return end
                        local extension = bytes:sub(1, 8) == "\137PNG\13\10\26\10" and "png" or bytes:sub(1, 2) == "\255\216" and "jpg"
                        if not extension then return end
                        return M.save("cover-" .. song.id .. "." .. extension, bytes)
                    end)
                    if alive and version == M.version and success and cover then M.cover.Image, M.miniCover.Image, M.detailCover.Image = cover, cover, cover end
                end)
            end
        end)
        if alive and version == M.version and not ok then
            M.loading, M.loaded = false, false
            M.status.Text = tostring(err)
            M.update()
        end
    end)
end
M.search = function()
    local query = M.input.Text:match("^%s*(.-)%s*$")
    if query == "" then M.searchStatus.Text = "请输入歌名、歌手或歌曲链接"; return end
    M.searchVersion = M.searchVersion + 1
    local version = M.searchVersion
    local id = query:match("^%d+$") or query:match("[?&]id=(%d+)") or query:match("/song/(%d+)")
    if id then
        M.searchButton.Text = "搜索"
        M.queue = { { id = id, name = "网易云 / " .. id, artist = "", duration = 0 } }
        M.playSong(M.queue[1], 1)
        return
    end
    M.searchButton.Text, M.searchStatus.Text = "搜索中", "正在搜索网易云…"
    task.spawn(function()
        local ok, songs = pcall(function()
            local params = "s=" .. HttpService:UrlEncode(query) .. "&type=1&limit=20&offset=0"
            local lastError
            for _, path in ipairs({ "/api/search/get?", "/api/cloudsearch/pc?", "/api/search/get/web?" }) do
                local success, data = pcall(M.json, path .. params)
                if not alive or version ~= M.searchVersion then return {} end
                if success and data.result and type(data.result.songs) == "table" then
                    local result = {}
                    for _, item in ipairs(data.result.songs) do result[#result + 1] = M.normalize(item) end
                    return result
                end
                lastError = success and "没有匹配歌曲" or tostring(data)
            end
            error(lastError, 0)
        end)
        if not alive or version ~= M.searchVersion then return end
        M.searchButton.Text = "搜索"
        if not ok then M.searchStatus.Text = "搜索失败：" .. tostring(songs); return end
        disconnect(M.resultConnections)
        for _, child in ipairs(M.results:GetChildren()) do child:Destroy() end
        M.queue, M.index = songs, 0
        M.results.CanvasSize = UDim2.fromOffset(0, #songs * 60)
        for index, song in ipairs(songs) do
            local row = round(make("TextButton", {
                Position = UDim2.fromOffset(0, (index - 1) * 60), Size = UDim2.fromOffset(316, 54),
                BackgroundColor3 = C.card, BorderSizePixel = 0, Text = "", AutoButtonColor = false,
                Interactable = state.open and state.musicVisible,
            }, M.results), 12)
            themed(row, "BackgroundColor3", "card")
            text(row, song.name, 13, 5, 290, 25, 14, C.white, true)
            text(row, song.artist, 13, 31, 290, 18, 11, C.muted)
            bind(row.Activated, function()
                M.queue = songs
                M.playSong(song, index)
            end, M.resultConnections)
        end
        M.searchStatus.Text = #songs > 0 and ("找到 " .. #songs .. " 首歌曲，点选播放") or "没有找到歌曲"
    end)
end
M.toggle = function()
    if M.loading then M.wants = not M.wants; M.status.Text = M.wants and "载入后播放" or "载入后暂停"; return end
    if not M.loaded then M.search(); return end
    if M.sound.IsPlaying then M.sound:Pause(); M.status.Text = "已暂停"
    else
        if M.ended then M.sound:Play(); M.ended = false else M.sound:Resume() end

local function attachHumanoid(component)
    if not alive or component == humanoid then return end
    disconnect(humanoidConnections)
    humanoid = component
    health.bound = true
    health.current, health.maximum = math.max(0, component.Health), component.MaxHealth
    island.demo = false
    refreshHealth(true)
    notify("ready", 0, false)
    bind(component.HealthChanged, function(value)
        local before = health.current
        health.current, health.maximum = math.max(0, value), component.MaxHealth
        local damage = before - health.current
        if damage > 0.001 then
            health.hits = health.hits + 1
            health.total = health.total + damage
            health.peak = math.max(health.peak, damage)
            updateStats()
            notify(health.current <= 0 and "death" or "damage", damage, false)
        end
        refreshHealth(false)
        if island.mode ~= "idle" and not island.demo then
            UI.alertDetail.Text = "当前生命 " .. number(health.current) .. " / " .. number(health.maximum)
        end
    end, humanoidConnections)
    bind(component:GetPropertyChangedSignal("MaxHealth"), function()
        health.maximum = component.MaxHealth
        refreshHealth(false)
    end, humanoidConnections)
end
local function detachHumanoid()
    disconnect(humanoidConnections)
    humanoid = nil
    health.bound = false
    refreshHealth(true)
end
local function attachCharacter(model)
    disconnect(characterConnections)
    detachHumanoid()
    character = model
    local component = model:FindFirstChildOfClass("Humanoid")
    if component then attachHumanoid(component) end
    bind(model.ChildAdded, function(child)
        if child:IsA("Humanoid") then attachHumanoid(child) end
    end, characterConnections)
    bind(model.ChildRemoved, function(child)
        if child == humanoid then detachHumanoid() end
    end, characterConnections)
end

local snowAlpha, fpsTime, fpsFrames = 0, 0, 0
local function render(dt)
    if not alive then return end
    local step = math.min(dt, 0.05)
    clock = clock + step
    local p = UI.progress.Value
    if state.motion then
        for _ = 1, 4 do
            local sub = step / 4
            islandVelocity = islandVelocity + (-180 * UI.islandJelly.Value - 13 * islandVelocity) * sub
            UI.islandJelly.Value = UI.islandJelly.Value + islandVelocity * sub
        end
    else
        UI.islandJelly.Value, islandVelocity = 0, 0
    end
    if math.abs(UI.islandJelly.Value) < 0.0001 and math.abs(islandVelocity) < 0.001 then
        UI.islandJelly.Value, islandVelocity = 0, 0
    end
    local islandJelly = UI.islandJelly.Value
    local islandWidth = UI.islandWidth.Value * (1 + islandJelly * 0.45)
    local islandHeight = UI.islandHeight.Value * (1 - islandJelly * 0.3)
    if islandWidth ~= islandGeometry.width or islandHeight ~= islandGeometry.height then
        UI.island.Size = UDim2.fromOffset(islandWidth, islandHeight)
        islandGeometry.width, islandGeometry.height = islandWidth, islandHeight
    end
    local visible = math.clamp(p, 0, 1)
    UI.shell.GroupTransparency = 1 - visible
    local travel = math.clamp(UI.travel.Value, 0, 1)
    local islandCenter = Vector2.new(UI.screen.AbsoluteSize.X / 2, Config.IslandTop + UI.islandHeight.Value * UI.islandScale.Scale / 2)
    local position = homePosition:Lerp(islandCenter, travel)
    UI.root.Position = UDim2.fromOffset(position.X, position.Y)
    UI.scale.Scale = fitScale * UI.zoom.Value
    UI.shell.Rotation = 0
    local musicPosition = musicHomePosition:Lerp(islandCenter, travel)
    UI.musicRoot.Position = UDim2.fromOffset(musicPosition.X, musicPosition.Y)
    UI.musicScale.Scale = fitScale * UI.zoom.Value * (0.94 + UI.musicPresence.Value * 0.06)
    UI.musicPanel.GroupTransparency = 1 - visible * UI.musicPresence.Value
    local musicVisible = UI.root.Visible and UI.musicPresence.Value > 0.002
    if state.musicVisible or not musicVisible then UI.musicRoot.Visible = musicVisible end
    local targetSnow = state.open and state.snow and visible or 0
    snowAlpha = snowAlpha + (targetSnow - snowAlpha) * (1 - math.exp(-step * 3))
    UI.snow.Visible = snowAlpha > 0.002
    local snowSize = UI.screen.AbsoluteSize
    local count = ({ 18, 32, Config.SnowCount })[state.density]
    local speed = ({ 0.65, 1, 1.6 })[state.snowSpeed]
    if snowAlpha > 0.002 then
        for i, flake in ipairs(flakes) do
            flake.node.Visible = i <= count
            if i <= count then
                flake.y = flake.y + step * flake.speed * speed / math.max(snowSize.Y, 1)
                if flake.y > 1.02 then flake.y = -0.03; flake.x = random:NextNumber() end
                flake.node.Position = UDim2.fromOffset(flake.x * snowSize.X + math.sin(clock * 0.35 + flake.phase) * flake.sway,
                    flake.y * snowSize.Y)
                flake.node.BackgroundTransparency = 1 - flake.opacity * snowAlpha
            end
        end
    end
    if state.open and state.page == 1 and state.motion then
        scene.model:PivotTo(CFrame.new(0, math.sin(clock * 0.7) * 0.12, 0) * CFrame.Angles(0.12, clock * 0.18, 0.12) * scene.pivot)
        scene.camera.CFrame = CFrame.lookAt(Vector3.new(tilt.X * 0.65, 0.8 - tilt.Y * 0.5, 10.2), Vector3.zero)
    end
    if islandExpanded and island.deadline <= os.clock() then
        island.mode, island.demo = "idle", false
        refreshHealth(false)
    end
    syncIsland()
    UI.alert.Visible = islandView == "alert" or UI.alert.GroupTransparency < 0.999
    UI.idle.Visible = islandView == "idle" or UI.idle.GroupTransparency < 0.999
    M.capsule.Visible = islandView == "music" or M.capsule.GroupTransparency < 0.999
    M.detail.Visible = islandView == "detail" or M.detail.GroupTransparency < 0.999
    local low = health.bound and health.current > 0 and health.current / math.max(health.maximum, 1) <= 0.25
    UI.idleDot.BackgroundTransparency = low and state.motion and (0.25 + math.sin(clock * 4) * 0.23) or 0
    M.tick(step, clock)
    UI.snowTitle.Text = state.snow and "一场慢雪" or "此刻，晴朗"
    UI.quickSnow.Text = state.snow and "落雪已开启" or "开启落雪"
    fpsTime, fpsFrames = fpsTime + dt, fpsFrames + 1
    if fpsTime >= 0.75 then
        UI.fps.Text = string.format("LIVE / %d FPS", math.floor(fpsFrames / fpsTime + 0.5))
        UI.timeText.Text = os.date("%H:%M")
        fpsTime, fpsFrames = 0, 0
    end
end
bind(RunService.RenderStepped, render)
local function cleanup()
    if not alive then return end
    alive = false
    drag = nil
    disconnect(connections)
    disconnect(characterConnections)
    disconnect(humanoidConnections)
    M.version, M.searchVersion = M.version + 1, M.searchVersion + 1
    if M.sound then M.sound:Stop(); M.sound:Destroy() end
    disconnect(M.resultConnections)
    local pending = {}
    for _, bag in pairs(tracks) do
        for _, animation in pairs(bag) do pending[animation] = true end
    end
    table.clear(tracks)
    for animation in pairs(pending) do animation:Cancel() end
end
destroy = function()
    if not alive then return end
    cleanup()
    UI.gui:Destroy()
end
bind(UI.gui.Destroying, cleanup)
